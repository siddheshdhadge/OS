#include <stdio.h>
#include <stdlib.h>

int cmp(const void *a, const void *b) {
    return (*(int *)a - *(int *)b);
}

int SSTF(int req[], int n, int head) {
    int total = 0, visited[n];
    for (int i = 0; i < n; i++) visited[i] = 0;
    for (int i = 0; i < n; i++) {
        int min = 1e9, idx = -1;
        for (int j = 0; j < n; j++) {
            if (!visited[j] && abs(req[j] - head) < min) {
                min = abs(req[j] - head);
                idx = j;
            }
        }
        visited[idx] = 1;
        total += min;
        head = req[idx];
    }
    return total;
}

int SCAN(int req[], int n, int head, int max) {
    int left[n], right[n], l = 0, r = 0, total = 0;
    for (int i = 0; i < n; i++)
        (req[i] < head) ? (left[l++] = req[i]) : (right[r++] = req[i]);
    qsort(left, l, sizeof(int), cmp);
    qsort(right, r, sizeof(int), cmp);
    if (r) total += abs(head - right[0]), head = right[r - 1];
    if (l) total += abs(head - left[0]);
    return total;
}

int C_LOOK(int req[], int n, int head, int max) {
    int left[n], right[n], l = 0, r = 0, total = 0;
    for (int i = 0; i < n; i++)
        (req[i] < head) ? (left[l++] = req[i]) : (right[r++] = req[i]);
    qsort(left, l, sizeof(int), cmp);
    qsort(right, r, sizeof(int), cmp);
    if (r) total += abs(head - right[r - 1]), head = right[r - 1];
    if (l) total += abs(head - left[0]);
    return total;
}

int main() {
    int n, head, max;
    printf("Enter number of requests: ");
    scanf("%d", &n);
    int req[n];
    printf("Enter requests: ");
    for (int i = 0; i < n; i++) scanf("%d", &req[i]);
    printf("Enter initial head: ");
    scanf("%d", &head);
    printf("Enter max track: ");
    scanf("%d", &max);

    printf("SSTF: Total head movement = %d\n", SSTF(req, n, head));
    printf("SCAN: Total head movement = %d\n", SCAN(req, n, head, max));
    printf("C-LOOK: Total head movement = %d\n", C_LOOK(req, n, head, max));
    return 0;
}
